BASIC CODE USAGE
═════════════════════════════════════════════════════════════════════

Simple Detection:
────────────────

import asyncio
from awaaz.src.pipeline.multilingual_predictions import get_multilingual_predictions

async def main():
    text = "नमस्ते। I need help। मेरे घर में पानी नहीं है।"
    predictions = await get_multilingual_predictions(text)
    
    print(f"Primary Language: {predictions['primary_language']}")
    print(f"Code-Switched: {predictions['is_code_switched']}")
    
    for seg in predictions['segments']:
        print(f"  {seg['language']}: {seg['text']}")

asyncio.run(main())


With STT Pipeline:
──────────────────

from awaaz.src.pipeline.stt import STTProcessor

async def main():
    stt = STTProcessor()
    await stt.load()
    
    # Transcribe
    result = await stt.transcribe("audio.wav")
    
    # Get per-sentence languages
    prediction = await stt.detect_sentence_languages(result.text)
    
    # Each segment has its language code for TTS!
    for seg in prediction.segments:
        print(f"{seg.language_code}: {seg.text}")

asyncio.run(main())
